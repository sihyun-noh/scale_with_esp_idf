idf_component_register(
    SRCS "dajilog_calibration_catalog.c"
    INCLUDE_DIRS "include" "${CMAKE_CURRENT_BINARY_DIR}"
    REQUIRES freertos
)

set(CALIBRATION_CSV "${CMAKE_CURRENT_LIST_DIR}/../../config/calibration/PRODUCT_CALIBRATION.csv")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${CALIBRATION_CSV}")
file(STRINGS "${CALIBRATION_CSV}" CALIBRATION_LINES ENCODING UTF-8)
list(POP_FRONT CALIBRATION_LINES CALIBRATION_HEADER)
if(NOT CALIBRATION_HEADER STREQUAL "schema_version,profile_id,product_code,product_lot,motor_channel,command_step,command_decivolt,discharge_g_min,valid,data_origin")
    message(FATAL_ERROR "Calibration CSV header is invalid")
endif()
list(LENGTH CALIBRATION_LINES CALIBRATION_ROW_COUNT)
if(CALIBRATION_ROW_COUNT GREATER 384 OR CALIBRATION_ROW_COUNT LESS 2)
    message(FATAL_ERROR "Calibration CSV must contain 2..384 rows")
endif()
set(CALIBRATION_DATA "")
foreach(CALIBRATION_LINE IN LISTS CALIBRATION_LINES)
    string(REPLACE "," ";" CALIBRATION_FIELDS "${CALIBRATION_LINE}")
    list(LENGTH CALIBRATION_FIELDS CALIBRATION_FIELD_COUNT)
    if(NOT CALIBRATION_FIELD_COUNT EQUAL 10)
        message(FATAL_ERROR "Calibration CSV row must have ten columns: ${CALIBRATION_LINE}")
    endif()
    list(GET CALIBRATION_FIELDS 0 CSV_VERSION)
    list(GET CALIBRATION_FIELDS 1 CSV_PROFILE)
    list(GET CALIBRATION_FIELDS 2 CSV_PRODUCT)
    list(GET CALIBRATION_FIELDS 3 CSV_LOT)
    list(GET CALIBRATION_FIELDS 4 CSV_CHANNEL)
    list(GET CALIBRATION_FIELDS 5 CSV_STEP)
    list(GET CALIBRATION_FIELDS 6 CSV_DECIVOLT)
    list(GET CALIBRATION_FIELDS 7 CSV_DISCHARGE)
    list(GET CALIBRATION_FIELDS 8 CSV_VALID)
    list(GET CALIBRATION_FIELDS 9 CSV_ORIGIN)
    if(NOT CSV_VERSION STREQUAL "1" OR
       NOT CSV_PROFILE MATCHES "^[A-Za-z0-9_.-]+$" OR
       NOT CSV_PRODUCT MATCHES "^[A-Za-z0-9_.-]+$" OR
       NOT CSV_LOT MATCHES "^[A-Za-z0-9_.-]+$" OR
       NOT CSV_CHANNEL MATCHES "^[ABCD]$" OR
       NOT CSV_STEP MATCHES "^[0-9]+$" OR
       NOT CSV_DECIVOLT MATCHES "^[0-9]+$" OR
       NOT CSV_DISCHARGE MATCHES "^[0-9]+$" OR
       NOT CSV_VALID MATCHES "^[01]$" OR
       NOT CSV_ORIGIN MATCHES "^(UNMEASURED|MEASURED)$")
        message(FATAL_ERROR "Calibration CSV row invalid or unverified origin missing: ${CALIBRATION_LINE}")
    endif()
    string(LENGTH "${CSV_PROFILE}" CSV_PROFILE_LENGTH)
    string(LENGTH "${CSV_PRODUCT}" CSV_PRODUCT_LENGTH)
    string(LENGTH "${CSV_LOT}" CSV_LOT_LENGTH)
    if(CSV_PROFILE_LENGTH GREATER_EQUAL 40 OR CSV_PRODUCT_LENGTH GREATER_EQUAL 32 OR
       CSV_LOT_LENGTH GREATER_EQUAL 32 OR CSV_STEP GREATER 24 OR
       CSV_DECIVOLT GREATER 120 OR CSV_DISCHARGE GREATER 1000000)
        message(FATAL_ERROR "Calibration CSV row exceeds supported bounds: ${CALIBRATION_LINE}")
    endif()
    if(CSV_ORIGIN STREQUAL "MEASURED")
        set(CSV_MEASURED 1)
    else()
        set(CSV_MEASURED 0)
    endif()
    string(APPEND CALIBRATION_DATA "    { \"${CSV_PROFILE}\", \"${CSV_PRODUCT}\", \"${CSV_LOT}\", '${CSV_CHANNEL}', ${CSV_STEP}U, ${CSV_DECIVOLT}U, ${CSV_DISCHARGE}U, ${CSV_VALID}, ${CSV_MEASURED} },\n")
endforeach()
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/dajilog_calibration_data.inc" "${CALIBRATION_DATA}")
